// zsprobe -- item 7 cause 1's design note (COORD 2026-11-05): which spellings of golib's zero-size gate does ILC fold
// at the slice indexer, with no class-constructor check left per element? A miniature of golib's slice<T> indexer and
// GoZeroSizeFacts<T> (the reflective initializer copied), each gate spelling its own indexer, each element type summed
// by its own non-inlined loop so the image shows whether the indexer inlined and what the gate cost. Not timed.
using System;
using System.Reflection;
using System.Runtime.CompilerServices;

interface IGoZeroSize { }
struct Empty : IGoZeroSize { }
struct Pair { public long A; public long B; }
struct Str { public byte[] B; public long L; }

static class Facts
{
    internal static bool Classify(Type type)
    {
        if (!type.IsValueType || type.IsPrimitive || type.IsEnum || type.IsPointer) return false;
        foreach (FieldInfo f in type.GetFields(BindingFlags.Instance | BindingFlags.Public | BindingFlags.NonPublic))
            if (!Classify(f.FieldType)) return false;
        return true;
    }
}
static class Facts<T>
{
    internal static readonly bool IsZeroSize = Facts.Classify(typeof(T));
    internal static readonly T[] Storage = IsZeroSize ? new T[1] : [];
}

readonly struct S<T>
{
    readonly T[] a; readonly nint low, len; readonly nuint nb;
    public S(T[] arr) { a = arr; low = 0; len = arr.Length; nb = 0; }

    // G0: today's gate.
    public ref T G0(nint i)
    {
        if (i < 0 || i >= len) throw new IndexOutOfRangeException();
        if (Facts<T>.IsZeroSize) return ref Zero();
        if (nb == 0) return ref a[low + i];
        return ref Native(i);
    }
    // G1: today's predicate's own non-reflective prefix as a pre-gate; the static is read only for other structs.
    public ref T G1(nint i)
    {
        if (i < 0 || i >= len) throw new IndexOutOfRangeException();
        if (typeof(T).IsValueType && !typeof(T).IsPrimitive && !typeof(T).IsEnum && Facts<T>.IsZeroSize) return ref Zero();
        if (nb == 0) return ref a[low + i];
        return ref Native(i);
    }
    // G2: an emitted marker, as a type test.
    public ref T G2(nint i)
    {
        if (i < 0 || i >= len) throw new IndexOutOfRangeException();
        if (typeof(IGoZeroSize).IsAssignableFrom(typeof(T))) return ref Zero();
        if (nb == 0) return ref a[low + i];
        return ref Native(i);
    }
    // G3: G1 plus "holds a reference", which Classify also answers non-zero (a reference field has size).
    public ref T G3(nint i)
    {
        if (i < 0 || i >= len) throw new IndexOutOfRangeException();
        if (typeof(T).IsValueType && !typeof(T).IsPrimitive && !typeof(T).IsEnum && !RuntimeHelpers.IsReferenceOrContainsReferences<T>() && Facts<T>.IsZeroSize) return ref Zero();
        if (nb == 0) return ref a[low + i];
        return ref Native(i);
    }
    [MethodImpl(MethodImplOptions.NoInlining)] static ref T Zero() => ref Facts<T>.Storage[0];
    [MethodImpl(MethodImplOptions.NoInlining)] ref T Native(nint i) => throw new NotSupportedException();
}

static class P
{
    [MethodImpl(MethodImplOptions.NoInlining)] static nuint U0(S<nuint> s, int n) { nuint t = 0; for (int i = 0; i < n; i++) t += s.G0(i); return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static nuint U1(S<nuint> s, int n) { nuint t = 0; for (int i = 0; i < n; i++) t += s.G1(i); return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static nuint U2(S<nuint> s, int n) { nuint t = 0; for (int i = 0; i < n; i++) t += s.G2(i); return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long Q0(S<Pair> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += s.G0(i).A; return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long Q1(S<Pair> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += s.G1(i).A; return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long Q2(S<Pair> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += s.G2(i).A; return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long U3(S<nuint> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += (long)s.G3(i); return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long Q3(S<Pair> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += s.G3(i).A; return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long R3(S<Str> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += s.G3(i).L; return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static long R1(S<Str> s, int n) { long t = 0; for (int i = 0; i < n; i++) t += s.G1(i).L; return t; }
    [MethodImpl(MethodImplOptions.NoInlining)] static int E3(S<Empty> s, int n) { int c = 0; for (int i = 0; i < n; i++) if (Unsafe.AreSame(ref s.G3(i), ref s.G3(0))) c++; return c; }
    [MethodImpl(MethodImplOptions.NoInlining)] static int E0(S<Empty> s, int n) { int c = 0; for (int i = 0; i < n; i++) if (Unsafe.AreSame(ref s.G0(i), ref s.G0(0))) c++; return c; }
    [MethodImpl(MethodImplOptions.NoInlining)] static int E1(S<Empty> s, int n) { int c = 0; for (int i = 0; i < n; i++) if (Unsafe.AreSame(ref s.G1(i), ref s.G1(0))) c++; return c; }
    [MethodImpl(MethodImplOptions.NoInlining)] static int E2(S<Empty> s, int n) { int c = 0; for (int i = 0; i < n; i++) if (Unsafe.AreSame(ref s.G2(i), ref s.G2(0))) c++; return c; }

    static int Main(string[] args)
    {
        int n = 8 + args.Length;
        var u = new S<nuint>(new nuint[n]); var q = new S<Pair>(new Pair[n]); var e = new S<Empty>(new Empty[n]);
        for (int i = 0; i < n; i++) { u.G0(i) = (nuint)i; q.G0(i).A = i; }
        long q0 = Q0(q, n), q1 = Q1(q, n), q2 = Q2(q, n);
        Console.WriteLine(q0); Console.WriteLine(q1); Console.WriteLine(q2);
        long d = 0; for (int i = 0; i < n; i++) d += q.G0(i).A; Console.WriteLine("direct " + d);
        var r = new S<Str>(new Str[n]); for (int i = 0; i < n; i++) r.G0(i).L = i;
        Console.WriteLine($"G3: {U3(u, n)} {Q3(q, n)} {R3(r, n)} {R1(r, n)} {E3(e, n)}");
        Console.WriteLine($"{U0(u, n)} {U1(u, n)} {U2(u, n)} {Q0(q, n)} {Q1(q, n)} {Q2(q, n)} {E0(e, n)} {E1(e, n)} {E2(e, n)}");
        return 0;
    }
}
