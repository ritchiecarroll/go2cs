#!/usr/bin/env python3
# Scratch, never committed: emits Proto384.cs, a straight-line six-limb Montgomery multiply (CIOS) with fiat's p384Mul
# contract (R = 2^384, m' = -p^-1 mod 2^64 = 0x100000001, output fully reduced), in two product spellings:
#   Big  - Math.BigMul;   Four - Go's portable four-32x32-product Mul64.
# Constant time in H's shape: straight line, bitwise carries (math/bits' Add64 formula), a mask-selected final
# subtraction. The reduction keeps its multiplies: P-384's prime is not Mersenne.
N = 6
P = [0x00000000ffffffff, 0xffffffff00000000, 0xfffffffffffffffe, 0xffffffffffffffff, 0xffffffffffffffff, 0xffffffffffffffff]
I = "        "

def body(prod):
    o = []
    w = o.append
    w(I + " ".join(f"ulong a{j} = arg1[{j}];" for j in range(N)))
    w(I + " ".join(f"ulong b{j} = arg2[{j}];" for j in range(N)))
    w(I + "ulong " + ", ".join(f"t{j} = 0" for j in range(N + 2)) + ";")
    w(I + "ulong hi, lo, c, k, s, m;")
    for i in range(N):
        w(f"{I}// round {i}: t += a * b{i}")
        w(f"{I}c = 0;")
        for j in range(N):
            w(f"{I}hi = {prod}(a{j}, b{i}, out lo);")
            w(f"{I}s = lo + t{j}; k = ((lo & t{j}) | ((lo | t{j}) & ~s)) >> 63; lo = s; hi += k;")
            w(f"{I}s = lo + c; k = ((lo & c) | ((lo | c) & ~s)) >> 63; t{j} = s; c = hi + k;")
        w(f"{I}s = t6 + c; k = ((t6 & c) | ((t6 | c) & ~s)) >> 63; t6 = s; t7 = k;")
        w(f"{I}// t = (t + m*p) / 2^64, m = t0 * m'")
        w(f"{I}m = t0 * 0x100000001UL;")
        w(f"{I}hi = {prod}(m, 0x{P[0]:X}UL, out lo);")
        w(f"{I}s = lo + t0; k = ((lo & t0) | ((lo | t0) & ~s)) >> 63; c = hi + k;")
        for j in range(1, N):
            w(f"{I}hi = {prod}(m, 0x{P[j]:X}UL, out lo);")
            w(f"{I}s = lo + t{j}; k = ((lo & t{j}) | ((lo | t{j}) & ~s)) >> 63; lo = s; hi += k;")
            w(f"{I}s = lo + c; k = ((lo & c) | ((lo | c) & ~s)) >> 63; t{j - 1} = s; c = hi + k;")
        w(f"{I}s = t6 + c; k = ((t6 & c) | ((t6 | c) & ~s)) >> 63; t5 = s; t6 = t7 + k; t7 = 0;")
    w(f"{I}// t < 2p: subtract p once with a borrow chain, keep d when no borrow or t6 carries.")
    w(f"{I}ulong borrow = 0;")
    for j in range(N):
        pj = f"0x{P[j]:X}UL"
        w(f"{I}ulong d{j} = t{j} - {pj} - borrow; borrow = ((~t{j} & {pj}) | (~(t{j} ^ {pj}) & d{j})) >> 63;")
    w(f"{I}ulong keepD = 0UL - ((borrow ^ 1) | t6);")
    for j in range(N):
        w(f"{I}out1[{j}] = (d{j} & keepD) | (t{j} & ~keepD);")
    return "\n".join(o)

print("""// GENERATED by gen_p384_proto.py (scratch, never committed).

    [MethodImpl(MethodImplOptions.AggressiveInlining)]
    static ulong Four(ulong x, ulong y, out ulong lo)
    {

        ulong w1 = t & 0xFFFFFFFF, w2 = t >> 32;
        w1 += x0 * y1;
        lo = x * y;
        return x1 * y1 + w2 + (w1 >> 32);
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    public static void MulBig(ulong[] out1, ulong[] arg1, ulong[] arg2)
    {
""" + body("Big") + """
    }

    [MethodImpl(MethodImplOptions.NoInlining)]
    public static void MulFour(ulong[] out1, ulong[] arg1, ulong[] arg2)
    {
