    s = t6 + c; k = ((t6 & c) | ((t6 | c) & ~s)) >> 63; t5 = s; t6 = t7 + k; t7 = 0;
    // t < 2p: subtract p once with a borrow chain, keep d when no borrow or t6 carries.
    ulong borrow = 0;
    ulong d0 = t0 - 0xFFFFFFFFUL - borrow; borrow = ((~t0 & 0xFFFFFFFFUL) | (~(t0 ^ 0xFFFFFFFFUL) & d0)) >> 63;
    ulong d1 = t1 - 0xFFFFFFFF00000000UL - borrow; borrow = ((~t1 & 0xFFFFFFFF00000000UL) | (~(t1 ^ 0xFFFFFFFF00000000UL) & d1)) >> 63;
    ulong d2 = t2 - 0xFFFFFFFFFFFFFFFEUL - borrow; borrow = ((~t2 & 0xFFFFFFFFFFFFFFFEUL) | (~(t2 ^ 0xFFFFFFFFFFFFFFFEUL) & d2)) >> 63;
    ulong d3 = t3 - 0xFFFFFFFFFFFFFFFFUL - borrow; borrow = ((~t3 & 0xFFFFFFFFFFFFFFFFUL) | (~(t3 ^ 0xFFFFFFFFFFFFFFFFUL) & d3)) >> 63;
    ulong d4 = t4 - 0xFFFFFFFFFFFFFFFFUL - borrow; borrow = ((~t4 & 0xFFFFFFFFFFFFFFFFUL) | (~(t4 ^ 0xFFFFFFFFFFFFFFFFUL) & d4)) >> 63;
    ulong d5 = t5 - 0xFFFFFFFFFFFFFFFFUL - borrow; borrow = ((~t5 & 0xFFFFFFFFFFFFFFFFUL) | (~(t5 ^ 0xFFFFFFFFFFFFFFFFUL) & d5)) >> 63;
    ulong keepD = 0UL - ((borrow ^ 1) | t6);
    out1[0] = (d0 & keepD) | (t0 & ~keepD);
    out1[1] = (d1 & keepD) | (t1 & ~keepD);
    out1[2] = (d2 & keepD) | (t2 & ~keepD);
    out1[3] = (d3 & keepD) | (t3 & ~keepD);
    out1[4] = (d4 & keepD) | (t4 & ~keepD);
    out1[5] = (d5 & keepD) | (t5 & ~keepD);
}
internal static void p384Square(ref p384MontgomeryDomainFieldElement out1, ref p384MontgomeryDomainFieldElement arg1) {
    p384Mul(ref out1, ref arg1, ref arg1);
}
[MethodImpl(MethodImplOptions.AggressiveInlining)]
private static ulong p384Mul64(ulong x, ulong y, out ulong lo) {
    ulong x0 = (uint)x, x1 = x >> 32, y0 = (uint)y, y1 = y >> 32;
    ulong p00 = x0 * y0, p01 = x0 * y1, p10 = x1 * y0, p11 = x1 * y1;
    ulong mid = (p00 >> 32) + (uint)p01 + (uint)p10;
    lo = (mid << 32) | (uint)p00;
    return p11 + (p01 >> 32) + (p10 >> 32) + (mid >> 32);
}
