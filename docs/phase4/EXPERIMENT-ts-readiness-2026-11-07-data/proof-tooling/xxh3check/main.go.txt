// xxh3check: native zeebo/xxh3 v1.1.0 values over fixed inputs, one line per input:
//   <name> <len> h128=<hi>:<lo> h64=<v> b128=<hi>:<lo> (b128 is Hash128 over the same bytes as []byte)
// The C# twin prints the same lines from System.IO.Hashing; the two outputs must be byte-identical.
package main

import (
	"fmt"
	"strings"

	"github.com/zeebo/xxh3"
)

func inputs() [][2]string {
	in := [][2]string{{"erased", "-"}, {"canonical", "*"}, {"base", "#"}, {"inner", "<"}, {"empty", ""}}
	for _, n := range []int{1, 3, 4, 8, 9, 16, 17, 128, 129, 240, 241, 1024, 4097} {
		var b strings.Builder
		for i := 0; i < n; i++ {
			b.WriteByte(byte('a' + (i*7+n)%26))
		}
		in = append(in, [2]string{fmt.Sprintf("len%d", n), b.String()})
	}
	return append(in, [2]string{"utf8", "héllo wörld ж 世界"})
}

func main() {
	for _, kv := range inputs() {
		s := kv[1]
		h := xxh3.HashString128(s)
		b := xxh3.Hash128([]byte(s))
		fmt.Printf("%s %d h128=%016x:%016x h64=%016x b128=%016x:%016x\n", kv[0], len(s), h.Hi, h.Lo, xxh3.HashString(s), b.Hi, b.Lo)
	}
}
