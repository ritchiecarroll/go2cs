# tree, to reach the type-check closure's next layer. Not a gate result. Each file is backed up to <file>.diag-orig first;
# --restore puts every backup back.

root = sys.argv[1]
P = [

for rel, a, b, *cnt in P:
    want = cnt[0] if cnt else 1
    f = os.path.join(root, rel); o = f + ".diag-orig"
    if "--restore" in sys.argv:
        if os.path.exists(o):
            shutil.move(o, f); print("restored", rel)
        continue
    s = open(f, encoding="utf-8-sig", newline="").read()  # newline="": keep CRLF as it is
    crlf = "\r\n" in s
    aa, bb = (a.replace("\n", "\r\n"), b.replace("\n", "\r\n")) if crlf else (a, b)
    # already applied: the text is gone and its replacement is there (or is empty), or a later entry rewrote it again
    # after this one (C33 drops the line C31 fixed), so only the backup still carries it
    if s.count(aa) == 0 and (not bb or s.count(bb) >= want or
                             (os.path.exists(o) and aa in open(o, encoding="utf-8-sig", newline="").read())):
        print("already patched", rel); continue
    if s.count(aa) != want:
        sys.exit(f"{rel}: expected {want} match(es), found {s.count(aa)}")
    if not os.path.exists(o):
        shutil.copy2(f, o)
    bom = open(f, "rb").read(3) == b"\xef\xbb\xbf"
    open(f, "w", encoding="utf-8-sig" if bom else "utf-8", newline="").write(s.replace(aa, bb))
    print("patched", rel)
