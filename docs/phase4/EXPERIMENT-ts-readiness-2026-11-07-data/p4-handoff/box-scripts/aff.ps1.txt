# aff.ps1 <log> <argsfile> : pin self to CPUs 6-11 (0xFC0), run the NUL-separated command in argsfile (children inherit
# the affinity), log all output, exit with its rc. Args travel by file because pwsh -File re-parses -x:y tokens.
$Log = "$($args[0])"
$argv = [IO.File]::ReadAllText("$($args[1])").TrimEnd([char]0).Split([char]0)
[System.Diagnostics.Process]::GetCurrentProcess().ProcessorAffinity = [IntPtr]0xFC0
if ($env:AFF_DOTNET_ROOT) { $env:DOTNET_ROOT = $env:AFF_DOTNET_ROOT; $env:PATH = "$($env:AFF_DOTNET_ROOT);$($env:PATH)" }
$Exe = $argv[0]; $Rest = @(); if ($argv.Count -gt 1) { $Rest = $argv[1..($argv.Count-1)] }
& $Exe @Rest *> $Log
exit $LASTEXITCODE
