package main

import "fmt"

// TS site: internal/checker grammarchecks.go checkGrammarBreakOrContinueStatement (C41): a pointer parameter that is
// also read through (node.Label()) and assigned to a pointer variable with an EXPLICIT type, `var current *Node = node`.
type Node struct {
	Kind   int
	Parent *Node
}

func (n *Node) Label() int { return n.Kind }

func depth(node *Node) (int, int) {
	if node.Kind < 0 {
		return 0, 0
	}
	label := node.Label()
	var current *Node = node
	d := 0
	for current != nil {
		d++
		current = current.Parent
	}
	return label, d
}

func main() {
	root := &Node{Kind: 1}
	leaf := &Node{Kind: 3, Parent: &Node{Kind: 2, Parent: root}}
	fmt.Println(depth(leaf)) // Go: 3 3
}
