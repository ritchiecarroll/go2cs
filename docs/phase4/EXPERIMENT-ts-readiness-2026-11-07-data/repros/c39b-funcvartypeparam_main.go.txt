package main

import "fmt"

// C39 variant (RED at 38caf3558a): a func-typed VARIABLE as the argument of the type parameter T. It is boxed by
// OrTypedNilFunc (object), so C# infers T = object.
func IfElse[T any](b bool, whenTrue T, whenFalse T) T {
	if b {
		return whenTrue
	}
	return whenFalse
}

type gen struct{ used map[string]bool }

func (g *gen) isUnique(name string, private bool) bool { return !g.used[name] }

func makeName(base string, check func(name string, private bool) bool) string {
	if check == nil || check(base, false) {
		return base
	}
	return base + "_1"
}

func main() {
	g := &gen{used: map[string]bool{"x": true}}
	var f func(name string, private bool) bool = g.isUnique
	fmt.Println(makeName("x", IfElse(true, f, nil)), makeName("x", IfElse(false, f, nil))) // Go: x_1 x
}
