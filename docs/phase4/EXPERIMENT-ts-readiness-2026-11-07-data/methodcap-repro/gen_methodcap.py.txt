# gen_methodcap.py <out dir> <N> <M> [K] : writes a Go module that reproduces the CLR per-type method cap in go2cs output.
#
# One struct, Base, with N pointer-receiver methods, embedded directly in M struct types T0..T{M-1}. Go promotes Base's N
# methods into each Ti's method set (the shape of typescript-go's internal/ast, where ~214 node types embed Node via
# NodeDefault). go2cs-gen's TypeGenerator emits each promoted method as a pair of extension forwarders
# (`this ref Ti` and `this ж<Ti>`) into the one package class, so that class carries 2*N*M + 2*N + 5 (+ K) methods, measured.
# CoreCLR refuses to load a type past its slot limit (TypeLoadException "contains more methods than the current
# implementation allows"), so the converted program fails before main once that count passes the threshold. The
# threshold is MEASURED with this generator (README.md), not taken from the runtime's source constant.
#
# K (optional, default 0) adds K plain package funcs, each one more method on the package class. It varies the count by
# one-method steps at a fixed N and M shape; K=0 writes exactly the source the earlier runs used.
#
# main calls a promoted method on the first and last embedding types and prints a fixed line, so the Go oracle output
# is deterministic and the converted program's output can be compared byte for byte.
import os, sys

out, n, m = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
k = int(sys.argv[4]) if len(sys.argv) > 4 else 0   # K plain package funcs (each one more method on the package class)
os.makedirs(out, exist_ok=True)
name = f"methodcap-n{n}-m{m}" + (f"-k{k}" if k else "")
with open(os.path.join(out, "go.mod"), "w", newline="\n") as f:
    f.write(f"module repro/{name}\n\ngo 1.26\n")
L = ["// Code generated by gen_methodcap.py; DO NOT EDIT.", f"// N={n} methods on Base, embedded in M={m} types.", "",
     "package main", "", 'import "fmt"', "", "type Base struct{ v int }", ""]
L += [f"func (b *Base) M{i}() int {{ return b.v + {i} }}" for i in range(n)]
L += [""] + [f"type T{j} struct{{ Base }}" for j in range(m)]
if k:
    L += [""] + [f"func F{i}() int {{ return {i} }}" for i in range(k)]
L += ["", "func main() {",
      "\tfirst := &T0{Base{1}}",
      f"\tlast := &T{m - 1}{{Base{{2}}}}",
      f'\tfmt.Println("N", {n}, "M", {m}, "promoted forwarders 2*N*M", {2 * n * m}, "calls", first.M0(), last.M{n - 1}()' + (f', "K", {k}, "F", F{k - 1}()' if k else "") + ')',
      "}", ""]
with open(os.path.join(out, "main.go"), "w", newline="\n") as f:
    f.write("\n".join(L))
print(f"{out}: N={n} M={m} K={k} 2*N*M={2 * n * m}")
