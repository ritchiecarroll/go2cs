# gen_methodcap.py <out dir> <N> <M> : writes a Go module that reproduces the CLR per-type method cap in go2cs output.
#
# One struct, Base, with N pointer-receiver methods, embedded directly in M struct types T0..T{M-1}. Go promotes Base's N
# methods into each Ti's method set (the shape of typescript-go's internal/ast, where ~214 node types embed Node via
# NodeDefault). go2cs-gen's TypeGenerator emits each promoted method as a pair of extension forwarders
# (`this ref Ti` and `this ж<Ti>`) into the one package class, so that class carries about 2*N*M methods. CoreCLR refuses
# to load a type with more than 65,535 methods (TypeLoadException "contains more methods than the current implementation
# allows"), so the converted program fails before main once 2*N*M plus the class's other methods passes 65,535.
#
# main calls a promoted method on the first and last embedding types and prints a fixed line, so the Go oracle output
# is deterministic and the converted program's output can be compared byte for byte.
import os, sys

out, n, m = sys.argv[1], int(sys.argv[2]), int(sys.argv[3])
os.makedirs(out, exist_ok=True)
name = f"methodcap-n{n}-m{m}"
with open(os.path.join(out, "go.mod"), "w", newline="\n") as f:
    f.write(f"module repro/{name}\n\ngo 1.26\n")
L = ["// Code generated by gen_methodcap.py; DO NOT EDIT.", f"// N={n} methods on Base, embedded in M={m} types.", "",
     "package main", "", 'import "fmt"', "", "type Base struct{ v int }", ""]
L += [f"func (b *Base) M{i}() int {{ return b.v + {i} }}" for i in range(n)]
L += [""] + [f"type T{j} struct{{ Base }}" for j in range(m)]
L += ["", "func main() {",
      "\tfirst := &T0{Base{1}}",
      f"\tlast := &T{m - 1}{{Base{{2}}}}",
      f'\tfmt.Println("N", {n}, "M", {m}, "promoted forwarders 2*N*M", {2 * n * m}, "calls", first.M0(), last.M{n - 1}())',
      "}", ""]
with open(os.path.join(out, "main.go"), "w", newline="\n") as f:
    f.write("\n".join(L))
print(f"{out}: N={n} M={m} 2*N*M={2 * n * m}")
