  mkdir -p "$HOME/sdk" && cp -a "$(go env GOMODCACHE)/golang.org/toolchain@v0.0.1-go1.26.9.linux-amd64" "$HOME/sdk/go1.26.9"
  chmod -R u+w "$HOME/sdk/go1.26.9"

`tar -C "$HOME/sdk" ...` with a rename to go1.26.9.
